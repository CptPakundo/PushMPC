# PushMPC-Setup.exe: the single-file installer (see src/setup.cpp). Shares the updater's helpers (../updater/src/common).
#   -DPUSHMPC_PAYLOAD=<setup-payload.bin>   the filter and core components, written by "pushmpc-install.ps1 package"
#                                           (without it: a developer build that can only repair, update or uninstall)
cmake_minimum_required(VERSION 3.24)
project(PushMpcSetup CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(PUSHMPC_PAYLOAD "" CACHE FILEPATH "setup payload")
set(PUSHMPC_SETUP_SRC "${CMAKE_CURRENT_SOURCE_DIR}")
set(UPD "${CMAKE_CURRENT_SOURCE_DIR}/../updater/src")
if(PUSHMPC_PAYLOAD)
  string(REPLACE "\\" "/" P "${PUSHMPC_PAYLOAD}")
  set(PUSHMPC_PAYLOAD_LINE "PAYLOAD RCDATA \"${P}\"")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PUSHMPC_PAYLOAD}")
else()
  set(PUSHMPC_PAYLOAD_LINE "")
endif()
configure_file(res/setup.rc.in ${CMAKE_CURRENT_BINARY_DIR}/setup.rc @ONLY)
add_executable(PushMpcSetup WIN32 src/setup.cpp ${UPD}/common/pm_common.cpp ${UPD}/common/pm_net.cpp ${UPD}/common/pm_release.cpp ${CMAKE_CURRENT_BINARY_DIR}/setup.rc)
if(PUSHMPC_PAYLOAD)
  set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/setup.rc PROPERTIES OBJECT_DEPENDS "${PUSHMPC_PAYLOAD}")
endif()
target_compile_definitions(PushMpcSetup PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(PushMpcSetup PRIVATE /W3 /EHsc /O2 /utf-8 /Z7)
target_include_directories(PushMpcSetup PRIVATE ${UPD} ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_options(PushMpcSetup PRIVATE /DEBUG /OPT:REF /PDBALTPATH:%_PDB% /MANIFEST:NO)
set_target_properties(PushMpcSetup PROPERTIES OUTPUT_NAME "PushMPC-Setup" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/bin")
