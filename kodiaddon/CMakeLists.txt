# PushMPC for Kodi: inputstream.pushmpc, one binary add-on that is both a Kodi inputstream (it claims local video files
# by extension and demuxes them with FFmpeg) and a Kodi videocodec (Kodi hands it the video packets of those files; it
# decodes them, runs PushMPC's engine and returns k pictures per source frame with their own timestamps).
#
#   -DKODI_DEVKIT=<xbmc>/xbmc/addons/kodi-dev-kit/include     the dev kit of the Kodi major to build for (21 Omega,
#                                                             22 Piers); one build per major, the API versions differ
#   -DFFMPEG_ROOT=<dir with include/ and lib/>                FFmpeg (static on Linux/macOS/Android, shared on Windows)
#
# Output: <build>/addon/inputstream.pushmpc/ (addon.xml, library, resources), ready to zip.
cmake_minimum_required(VERSION 3.20)
project(inputstream_pushmpc C CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/msvc-loop-guard.cmake")   # MSVC loop optimiser bug guard (/d2NoLUR)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_C_VISIBILITY_PRESET hidden)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(KODI_DEVKIT "" CACHE PATH "kodi-dev-kit/include of the target Kodi")
set(FFMPEG_ROOT "" CACHE PATH "FFmpeg prefix (include/, lib/)")
# the add-on version: PushMPC's own (filter/src/version.h), so Kodi sees a newer zip as an update

string(REGEX MATCH "\"([0-9]+[.][0-9]+[.][0-9]+)" _m "${_pmv}")

# library loaded at run time through volk) first, PushFPS's portable core on the CPU (vsplugin/engines/portable behind
# pm_engine.h) as the fallback. The GPU engine needs the Vulkan-Headers and volk that vsplugin/engines/gpu/gpu-pins.json
# pins (fetched at configure time with their SHA-256, or given as PMVS_VULKAN_HEADERS_DIR / PMVS_VOLK_DIR).
option(PM_GPU "build the portable GPU engine" ON)

set(PMVS_VOLK_DIR "" CACHE PATH "a volk checkout")

  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
  add_compile_options(/EHsc /utf-8)
endif()

target_include_directories(pm_engine_portable PUBLIC src/engine)
target_link_libraries(pm_engine_portable PUBLIC pm_portable)

set(PM_GPU_ON OFF)
if(PM_GPU)
  if(NOT PMVS_VULKAN_HEADERS_DIR OR NOT PMVS_VOLK_DIR)
    file(READ "${VS}/engines/gpu/gpu-pins.json" _pins)
    include(FetchContent)
    string(JSON _vh_url GET "${_pins}" vulkanHeaders url)
    string(JSON _vh_sha GET "${_pins}" vulkanHeaders sha256)
    string(JSON _volk_url GET "${_pins}" volk url)
    string(JSON _volk_sha GET "${_pins}" volk sha256)
    FetchContent_Declare(pm_vulkan_headers URL "${_vh_url}" URL_HASH SHA256=${_vh_sha} DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_Declare(pm_volk URL "${_volk_url}" URL_HASH SHA256=${_volk_sha} DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_GetProperties(pm_vulkan_headers)
    if(NOT pm_vulkan_headers_POPULATED)
      FetchContent_Populate(pm_vulkan_headers)
    endif()
    FetchContent_GetProperties(pm_volk)
    if(NOT pm_volk_POPULATED)
      FetchContent_Populate(pm_volk)
    endif()
    set(PMVS_VULKAN_HEADERS_DIR "${pm_vulkan_headers_SOURCE_DIR}")
    set(PMVS_VOLK_DIR "${pm_volk_SOURCE_DIR}")
  endif()
  if(EXISTS "${PMVS_VULKAN_HEADERS_DIR}/include/vulkan/vulkan_core.h" AND EXISTS "${PMVS_VOLK_DIR}/volk.c")
    set(PM_GPU_ON ON)
  endif()
endif()

# ---- Linux x86_64 with NVIDIA: the nvl engine (vsplugin/engines/nvidia-linux: VFG, NVIDIA optical flow, RTX VSR, the
# cut guard) through CUDA and NVIDIA's Video Effects runtime, both loaded at run time; nvcc is needed at build time only
# (the kernels' fatbin; the GPU cut guard's fatbins from slangc when present, else the CPU guard). Without nvcc the
# add-on is built without it and the portable GPU engine runs on NVIDIA too.
set(PM_NVL_DEFAULT OFF)
if(UNIX AND NOT APPLE AND NOT ANDROID AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  set(PM_NVL_DEFAULT ON)
endif()
option(PM_NVL "the NVIDIA engine on Linux (needs nvcc at build time)" ${PM_NVL_DEFAULT})
if(PM_NVL)
  find_program(PM_NVCC nvcc HINTS "$ENV{CUDA_PATH}/bin" /usr/local/cuda/bin)
  if(NOT PM_NVCC)
    message(WARNING "nvl: no nvcc: the add-on is built without the NVIDIA engine")
    set(PM_NVL OFF)
  endif()
endif()
if(PM_NVL)
  set(NVL "${VS}/engines/nvidia-linux")
  set(_gencode "-gencode;arch=compute_75,code=sm_75;-gencode;arch=compute_86,code=sm_86;-gencode;arch=compute_89,code=sm_89;-gencode;arch=compute_120,code=sm_120;-gencode;arch=compute_75,code=compute_75;-gencode;arch=compute_120,code=compute_120")
  set(_fb "${CMAKE_BINARY_DIR}/nvl_kernels.fatbin")
  add_custom_command(OUTPUT "${_fb}"
    COMMAND "${PM_NVCC}" -fatbin -O3 -std=c++17 -I "${NVL}" ${_gencode} -o "${_fb}" "${NVL}/nvl_kernels.cu"
    DEPENDS "${NVL}/nvl_kernels.cu" "${NVL}/nvl_colour.h" VERBATIM COMMAND_EXPAND_LISTS)
  add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/nvl_kernels_fatbin.cpp"
    COMMAND "${CMAKE_COMMAND}" -DIN=${_fb} -DOUT=${CMAKE_BINARY_DIR}/nvl_kernels_fatbin.cpp -P "${NVL}/bin2cpp.cmake"
    DEPENDS "${_fb}" "${NVL}/bin2cpp.cmake" VERBATIM)
  set(_guard "")
  find_program(PM_SLANGC slangc HINTS "$ENV{SLANG_DIR}/bin")
  if(PM_SLANGC)
    get_filename_component(_sroot "${PM_SLANGC}" DIRECTORY)
    foreach(g sc fm)
      set(src "${NVL}/shaders/VfgShotChange.hlsl")
      if(g STREQUAL "fm")
        set(src "${NVL}/shaders/VfgFourierMellin.hlsl")
      endif()
      set(gfb "${CMAKE_BINARY_DIR}/nvl_guard_${g}.fatbin")
      add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/nvl_guard_${g}.cu"
        COMMAND "${PM_SLANGC}" -lang slang "${src}" -entry main -stage compute -target cuda -o "${CMAKE_BINARY_DIR}/nvl_guard_${g}.cu" DEPENDS "${src}" VERBATIM)
      add_custom_command(OUTPUT "${gfb}"
        COMMAND "${PM_NVCC}" -fatbin -O3 -std=c++17 -include cuda_fp16.h -I "${_sroot}/../include" ${_gencode} -o "${gfb}" "${CMAKE_BINARY_DIR}/nvl_guard_${g}.cu"
        DEPENDS "${CMAKE_BINARY_DIR}/nvl_guard_${g}.cu" VERBATIM COMMAND_EXPAND_LISTS)
      add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/nvl_guard_${g}_fatbin.cpp"
        COMMAND "${CMAKE_COMMAND}" -DIN=${gfb} -DOUT=${CMAKE_BINARY_DIR}/nvl_guard_${g}_fatbin.cpp -DNAME=nvl_guard_${g}_fatbin -P "${NVL}/bin2cpp.cmake"
        DEPENDS "${gfb}" "${NVL}/bin2cpp.cmake" VERBATIM)
      list(APPEND _guard "${CMAKE_BINARY_DIR}/nvl_guard_${g}_fatbin.cpp")
    endforeach()
  endif()
  add_library(pm_nvl STATIC "${NVL}/nvl_loader.cpp" "${NVL}/nvl_engine.cpp" "${NVL}/nvl_ampere.cpp" "${NVL}/nvl_selftest.cpp"
              "${NVL}/nvl_gpuguard.cpp" "${NVL}/nvl_nvof.cpp" "${CMAKE_BINARY_DIR}/nvl_kernels_fatbin.cpp" ${_guard})
  if(_guard)
    target_compile_definitions(pm_nvl PRIVATE PMVS_NVL_GPU_GUARD)
  endif()
  target_include_directories(pm_nvl PUBLIC "${NVL}" "${VS}/src")
  target_link_libraries(pm_nvl PUBLIC ${CMAKE_DL_LIBS} Threads::Threads)
  target_compile_definitions(pm_kodi_engine PUBLIC PM_NVL)
  target_link_libraries(pm_kodi_engine PUBLIC pm_nvl)
  # the RTX 30 VFG gate (a child process; nvl_selftest.cpp finds it next to the add-on library)
  add_executable(pushmpc-nvvfx-selftest "${NVL}/nvl_selftest_main.cpp")
  target_link_libraries(pushmpc-nvvfx-selftest PRIVATE pm_nvl)
  target_link_options(pushmpc-nvvfx-selftest PRIVATE -static-libstdc++ -static-libgcc)
  message(STATUS "engines: + NVIDIA (nvl, Linux)")
endif()

  endif()

# Android: MediaCodec into AImageReader hardware buffers, imported into the GPU engine (src/android); the conversion
# shader is compiled by glslang at build time (-DPM_GLSLANG=<glslangValidator>); without it the zero-copy path is left
# out and FFmpeg's MediaCodec decoders (NDK mode) remain
set(PM_GLSLANG "" CACHE FILEPATH "glslangValidator for the Android conversion shader")
if(ANDROID)
  target_link_libraries(inputstream.pushmpc PRIVATE mediandk android log)
  if(PM_GLSLANG AND PM_GPU_ON)
    set(_aspv "${CMAKE_BINARY_DIR}/android_spv")
    file(MAKE_DIRECTORY "${_aspv}")
    add_custom_command(OUTPUT "${_aspv}/ahb2yuv_spv.h"
      COMMAND "${PM_GLSLANG}" -V --target-env vulkan1.2 --vn kSpv_ahb2yuv -o "${_aspv}/ahb2yuv_spv.h" "${CMAKE_CURRENT_SOURCE_DIR}/src/android/ahb2yuv.comp"
      DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/android/ahb2yuv.comp" COMMENT "glslang: ahb2yuv.comp" VERBATIM)
    target_sources(inputstream.pushmpc PRIVATE src/android/ahb_path.cpp "${_aspv}/ahb2yuv_spv.h")
    target_include_directories(inputstream.pushmpc PRIVATE "${_aspv}")
    target_compile_definitions(inputstream.pushmpc PRIVATE PM_AHB)
    # Android 8 APIs (AImageReader with usage, AImage hardware buffers): weak symbols, checked at run time
    set_source_files_properties(src/android/ahb_path.cpp PROPERTIES COMPILE_DEFINITIONS __ANDROID_UNAVAILABLE_SYMBOLS_ARE_WEAK__
                                COMPILE_OPTIONS "-Wno-unguarded-availability;-Wno-error=unguarded-availability")
    message(STATUS "Android: zero-copy MediaCodec path (hardware buffers into the GPU engine)")
  endif()
endif()
