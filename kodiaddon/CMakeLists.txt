# PushMPC for Kodi: inputstream.pushmpc, one binary add-on that is both a Kodi inputstream (it claims local video files
# by extension and demuxes them with FFmpeg) and a Kodi videocodec (Kodi hands it the video packets of those files; it
# decodes them, runs PushMPC's engine and returns k pictures per source frame with their own timestamps).
#
#   -DKODI_DEVKIT=<xbmc>/xbmc/addons/kodi-dev-kit/include     the dev kit of the Kodi major to build for (21 Omega,
#                                                             22 Piers); one build per major, the API versions differ
#   -DFFMPEG_ROOT=<dir with include/ and lib/>                FFmpeg (static on Linux/macOS/Android, shared on Windows)
#
# Output: <build>/addon/inputstream.pushmpc/ (addon.xml, library, resources), ready to zip.
cmake_minimum_required(VERSION 3.20)
project(inputstream_pushmpc C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_C_VISIBILITY_PRESET hidden)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(KODI_DEVKIT "" CACHE PATH "kodi-dev-kit/include of the target Kodi")
set(FFMPEG_ROOT "" CACHE PATH "FFmpeg prefix (include/, lib/)")
set(ADDON_VERSION "0.1.0" CACHE STRING "add-on version")
option(PM_BUILD_ADDON "build the add-on (needs KODI_DEVKIT and FFMPEG_ROOT)" ON)

set(VS "${CMAKE_CURRENT_SOURCE_DIR}/../vsplugin")
find_package(Threads REQUIRED)

# ---- the engines (kodi_engine.h): the portable GPU engine (vsplugin/engines/gpu, Vulkan 1.2 compute, the Vulkan
# library loaded at run time through volk) first, PushFPS's portable core on the CPU (vsplugin/engines/portable behind
# pm_engine.h) as the fallback. The GPU engine needs the Vulkan-Headers and volk that vsplugin/engines/gpu/gpu-pins.json
# pins (fetched at configure time with their SHA-256, or given as PMVS_VULKAN_HEADERS_DIR / PMVS_VOLK_DIR).
option(PM_GPU "build the portable GPU engine" ON)
set(PMVS_VULKAN_HEADERS_DIR "" CACHE PATH "a Vulkan-Headers checkout")
set(PMVS_VOLK_DIR "" CACHE PATH "a volk checkout")
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
  add_compile_options(/EHsc /utf-8)
endif()
set(PFP "${VS}/third_party/pushfps-portable")
add_library(pm_pushfps_portable STATIC "${PFP}/src/Motion.cpp" "${PFP}/src/Interpolate.cpp")
target_include_directories(pm_pushfps_portable PUBLIC "${PFP}/include")
target_link_libraries(pm_pushfps_portable PUBLIC Threads::Threads)
add_library(pm_portable STATIC "${VS}/engines/portable/portable_engine.cpp")
target_include_directories(pm_portable PUBLIC "${VS}/engines/portable" "${VS}/src")
target_link_libraries(pm_portable PUBLIC pm_pushfps_portable)
add_library(pm_engine_portable STATIC src/engine/engine_portable.cpp)
target_include_directories(pm_engine_portable PUBLIC src/engine)
target_link_libraries(pm_engine_portable PUBLIC pm_portable)

set(PM_GPU_ON OFF)
if(PM_GPU)
  if(NOT PMVS_VULKAN_HEADERS_DIR OR NOT PMVS_VOLK_DIR)
    file(READ "${VS}/engines/gpu/gpu-pins.json" _pins)
    include(FetchContent)
    string(JSON _vh_url GET "${_pins}" vulkanHeaders url)
    string(JSON _vh_sha GET "${_pins}" vulkanHeaders sha256)
    string(JSON _volk_url GET "${_pins}" volk url)
    string(JSON _volk_sha GET "${_pins}" volk sha256)
    FetchContent_Declare(pm_vulkan_headers URL "${_vh_url}" URL_HASH SHA256=${_vh_sha} DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_Declare(pm_volk URL "${_volk_url}" URL_HASH SHA256=${_volk_sha} DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_GetProperties(pm_vulkan_headers)
    if(NOT pm_vulkan_headers_POPULATED)
      FetchContent_Populate(pm_vulkan_headers)
    endif()
    FetchContent_GetProperties(pm_volk)
    if(NOT pm_volk_POPULATED)
      FetchContent_Populate(pm_volk)
    endif()
    set(PMVS_VULKAN_HEADERS_DIR "${pm_vulkan_headers_SOURCE_DIR}")
    set(PMVS_VOLK_DIR "${pm_volk_SOURCE_DIR}")
  endif()
  if(EXISTS "${PMVS_VULKAN_HEADERS_DIR}/include/vulkan/vulkan_core.h" AND EXISTS "${PMVS_VOLK_DIR}/volk.c")
    set(PM_GPU_ON ON)
  endif()
endif()
set(PM_ENGINE_SRC src/engine/kodi_engine.cpp)
add_library(pm_kodi_engine STATIC ${PM_ENGINE_SRC})
target_include_directories(pm_kodi_engine PUBLIC src/engine "${VS}/src")
target_link_libraries(pm_kodi_engine PUBLIC pm_engine_portable)
if(PM_GPU_ON)
  add_library(pm_gpu STATIC "${VS}/engines/gpu/vk_context.cpp" "${VS}/engines/gpu/gpu_engine.cpp" "${VS}/engines/gpu/subtitle_hold.cpp"
              "${PFP}/src/CrosshairHold.cpp" "${PMVS_VOLK_DIR}/volk.c")
  target_include_directories(pm_gpu PUBLIC "${VS}/engines/gpu" "${PMVS_VULKAN_HEADERS_DIR}/include" "${PMVS_VOLK_DIR}" "${VS}/third_party/nis"
                             PRIVATE "${VS}/engines/gpu/spv")
  target_compile_definitions(pm_gpu PUBLIC PMVS_GPU VK_NO_PROTOTYPES)
  target_link_libraries(pm_gpu PUBLIC pm_portable ${CMAKE_DL_LIBS})
  target_link_libraries(pm_kodi_engine PUBLIC pm_gpu)
  message(STATUS "engines: GPU (Vulkan) + CPU")
else()
  message(STATUS "engines: CPU only (no Vulkan-Headers / volk)")
endif()

# ---- unit tests with no Kodi and no FFmpeg: the cadence (which pictures, which timestamps) and the engine glue
add_executable(cadence_test tests/cadence_test.cpp)
target_include_directories(cadence_test PRIVATE src)
target_link_libraries(cadence_test PRIVATE pm_kodi_engine)
enable_testing()
add_test(NAME cadence_test COMMAND cadence_test)

if(NOT PM_BUILD_ADDON)
  return()
endif()
if(NOT EXISTS "${KODI_DEVKIT}/kodi/versions.h")
  message(FATAL_ERROR "KODI_DEVKIT must point at kodi-dev-kit/include (kodi/versions.h)")
endif()

# ---- addon.xml: the API versions come from this dev kit's versions.h, as Kodi's own AddonHelpers.cmake does it
file(STRINGS "${KODI_DEVKIT}/kodi/versions.h" _ver REGEX "^#define ADDON_(GLOBAL|INSTANCE)_VERSION_[A-Z]+(_MIN|_XML_ID)? ")
foreach(l ${_ver})
  string(REGEX MATCH "^#define (ADDON_[A-Z_]+) +\"([^\"]+)\"" _m "${l}")
  if(_m)
    set(${CMAKE_MATCH_1} "${CMAKE_MATCH_2}")
  endif()
endforeach()
set(ADDON_DEPENDS "")
foreach(part GLOBAL_VERSION_MAIN GLOBAL_VERSION_GENERAL GLOBAL_VERSION_FILESYSTEM GLOBAL_VERSION_TOOLS INSTANCE_VERSION_INPUTSTREAM INSTANCE_VERSION_VIDEOCODEC)
  if(NOT DEFINED ADDON_${part}_XML_ID)
    message(FATAL_ERROR "versions.h has no ADDON_${part}")
  endif()
  string(APPEND ADDON_DEPENDS "\n    <import addon=\"${ADDON_${part}_XML_ID}\" minversion=\"${ADDON_${part}_MIN}\" version=\"${ADDON_${part}}\"/>")
endforeach()
string(REGEX MATCH "^[0-9]+" KODI_API_MAJOR "${ADDON_INSTANCE_VERSION_INPUTSTREAM}")
if(ADDON_INSTANCE_VERSION_VIDEOCODEC VERSION_GREATER_EQUAL 2.2.0)
  set(PM_KODI22 1)
else()
  set(PM_KODI22 0)
endif()
message(STATUS "Kodi dev kit: inputstream API ${ADDON_INSTANCE_VERSION_INPUTSTREAM}, videocodec API ${ADDON_INSTANCE_VERSION_VIDEOCODEC}")

if(WIN32)
  set(ADDON_PLATFORM windx)
  set(ADDON_LIBRARY "inputstream.pushmpc.dll")
elseif(ANDROID)
  set(ADDON_PLATFORM android)
  set(ADDON_LIBRARY "libinputstream.pushmpc.so")
elseif(APPLE)
  set(ADDON_PLATFORM osx)
  set(ADDON_LIBRARY "inputstream.pushmpc.dylib")
else()
  set(ADDON_PLATFORM linux)
  set(ADDON_LIBRARY "inputstream.pushmpc.so.${ADDON_VERSION}")
endif()

set(OUT "${CMAKE_BINARY_DIR}/addon/inputstream.pushmpc")
configure_file(inputstream.pushmpc/addon.xml.in "${OUT}/addon.xml" @ONLY)
file(COPY inputstream.pushmpc/resources DESTINATION "${OUT}")
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE" DESTINATION "${OUT}")

# ---- FFmpeg
set(_ff_libs avformat avcodec swscale avutil)
set(FF_LINK "")
foreach(l ${_ff_libs})
  find_library(FF_${l} NAMES ${l} lib${l} PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH)
  if(NOT FF_${l})
    message(FATAL_ERROR "FFmpeg library ${l} not found in ${FFMPEG_ROOT}/lib")
  endif()
  list(APPEND FF_LINK "${FF_${l}}")
endforeach()

add_library(inputstream.pushmpc SHARED src/pushmpc_kodi.cpp src/engine/display_hz.c)
target_include_directories(inputstream.pushmpc PRIVATE "${KODI_DEVKIT}" "${FFMPEG_ROOT}/include" src src/engine)
target_compile_definitions(inputstream.pushmpc PRIVATE PM_ADDON_VERSION="${ADDON_VERSION}" PM_KODI_API_MAJOR=${KODI_API_MAJOR} PM_KODI22=${PM_KODI22})
target_link_libraries(inputstream.pushmpc PRIVATE pm_kodi_engine ${FF_LINK})
# the platform macros Kodi's own add-on build (AddonHelpers / KodiConfig) defines
if(WIN32)
  target_compile_definitions(inputstream.pushmpc PRIVATE TARGET_WINDOWS NOMINMAX _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN)
elseif(APPLE)
  target_compile_definitions(inputstream.pushmpc PRIVATE TARGET_POSIX TARGET_DARWIN TARGET_DARWIN_OSX)
elseif(ANDROID)
  target_compile_definitions(inputstream.pushmpc PRIVATE TARGET_POSIX TARGET_LINUX TARGET_ANDROID)
else()
  target_compile_definitions(inputstream.pushmpc PRIVATE TARGET_POSIX TARGET_LINUX)
endif()
set_target_properties(inputstream.pushmpc PROPERTIES PREFIX "" OUTPUT_NAME "inputstream.pushmpc"
                      LIBRARY_OUTPUT_DIRECTORY "${OUT}" RUNTIME_OUTPUT_DIRECTORY "${OUT}")
foreach(cfg Release Debug RelWithDebInfo)
  string(TOUPPER ${cfg} CFG)
  set_target_properties(inputstream.pushmpc PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${CFG} "${OUT}" RUNTIME_OUTPUT_DIRECTORY_${CFG} "${OUT}")
endforeach()
if(WIN32)
  target_link_libraries(inputstream.pushmpc PRIVATE user32 bcrypt ws2_32 secur32 ole32)
  # FFmpeg's DLLs go next to the add-on library (Kodi loads add-ons with the add-on folder on the DLL search path)
  file(GLOB _ff_dlls "${FFMPEG_ROOT}/bin/*.dll")
  foreach(d ${_ff_dlls})
    file(COPY "${d}" DESTINATION "${OUT}")
  endforeach()
elseif(APPLE)
  target_link_libraries(inputstream.pushmpc PRIVATE "-framework CoreGraphics" "-framework CoreFoundation" "-framework CoreVideo" "-framework VideoToolbox" "-framework CoreMedia" "-framework Security" z bz2 iconv)
else()
  # FFmpeg is linked statically and kept private: Kodi itself has FFmpeg loaded, so our copy must neither export nor
  # bind to its symbols (-Bsymbolic, --exclude-libs). The C++ runtime is static too, so the library only needs libc.
  set_target_properties(inputstream.pushmpc PROPERTIES SUFFIX ".so.${ADDON_VERSION}")
  target_link_options(inputstream.pushmpc PRIVATE -static-libstdc++ -static-libgcc -Wl,-Bsymbolic -Wl,--exclude-libs,ALL -Wl,--no-undefined)
  target_link_libraries(inputstream.pushmpc PRIVATE ${CMAKE_DL_LIBS} m)
endif()
