# The PushMPC engine module for FFmpeg's vf_pushmpc (pmeng.h): pushmpc_engine.dll (MSVC, /MT) on Windows,
# libpushmpc_engine.so / .dylib elsewhere. Engines: the portable CPU engine (vsplugin/engines/portable, every OS) and
# pushcore on D3D11 (Windows). Plus engine_test (CPU paths everywhere; D3D11 against the stub pushcore on WARP).
#   cmake -S mpvnative/engine -B build-engine -DCMAKE_BUILD_TYPE=Release && cmake --build build-engine --config Release
cmake_minimum_required(VERSION 3.21)
project(PushMpcEngine C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
enable_testing()
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(PFP "${ROOT}/vsplugin/third_party/pushfps-portable")
find_package(Threads REQUIRED)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
  add_compile_options(/W3 /EHsc /O2 /Zi /utf-8)
  add_link_options(/DEBUG /OPT:REF /PDBALTPATH:%_PDB%)
else()
  add_compile_options(-O2 -Wall -Wextra -Wno-unused-parameter)
endif()

add_library(pme_pushfps_portable STATIC "${PFP}/src/Motion.cpp" "${PFP}/src/Interpolate.cpp")
target_include_directories(pme_pushfps_portable PUBLIC "${PFP}/include")
target_link_libraries(pme_pushfps_portable PUBLIC Threads::Threads)
add_library(pme_portable STATIC "${ROOT}/vsplugin/engines/portable/portable_engine.cpp")
target_include_directories(pme_portable PUBLIC "${ROOT}/vsplugin/engines/portable" "${ROOT}/vsplugin/src")
target_link_libraries(pme_portable PUBLIC pme_pushfps_portable)

# ---- the portable GPU engine (vsplugin/engines/gpu) as the Vulkan zero-copy engine: Vulkan-Headers and volk pinned in
# gpu-pins.json (fetched with -DPME_FETCH_GPU_DEPS=ON), the engine's checked-in SPIR-V, and this module's own two shaders
# (import / export of FFmpeg frames) compiled by glslang (-DPME_GLSLANG=<glslangValidator>, else found on PATH).
# Without any of them the module has the CPU and D3D11 engines only, and Vulkan frames take the filter's copy fallback.
option(PME_VULKAN "build the Vulkan zero-copy engine when its dependencies are available" ON)
option(PME_FETCH_GPU_DEPS "download the pinned Vulkan-Headers and volk at configure time" OFF)
set(PME_VULKAN_HEADERS_DIR "" CACHE PATH "a Vulkan-Headers checkout")
set(PME_VOLK_DIR "" CACHE PATH "a volk checkout")
set(PME_GLSLANG "" CACHE FILEPATH "glslangValidator")
set(PME_VULKAN_ON OFF)
if(PME_VULKAN)
  file(READ "${ROOT}/vsplugin/engines/gpu/gpu-pins.json" _pins)
  if(PME_FETCH_GPU_DEPS AND (NOT PME_VULKAN_HEADERS_DIR OR NOT PME_VOLK_DIR))
    include(FetchContent)
    foreach(dep vulkanHeaders volk)
      string(JSON _u GET "${_pins}" ${dep} url)
      string(JSON _s GET "${_pins}" ${dep} sha256)
      FetchContent_Declare(pme_${dep} URL "${_u}" URL_HASH SHA256=${_s} DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
      FetchContent_GetProperties(pme_${dep})
      if(NOT pme_${dep}_POPULATED)
        FetchContent_Populate(pme_${dep})
      endif()
    endforeach()
    set(PME_VULKAN_HEADERS_DIR "${pme_vulkanheaders_SOURCE_DIR}" CACHE PATH "" FORCE)
    set(PME_VOLK_DIR "${pme_volk_SOURCE_DIR}" CACHE PATH "" FORCE)
  endif()
  if(NOT PME_GLSLANG)
    find_program(PME_GLSLANG_FOUND NAMES glslangValidator glslang)
    if(PME_GLSLANG_FOUND)
      set(PME_GLSLANG "${PME_GLSLANG_FOUND}")
    endif()
  endif()
  if(EXISTS "${PME_VULKAN_HEADERS_DIR}/include/vulkan/vulkan_core.h" AND EXISTS "${PME_VOLK_DIR}/volk.c" AND PME_GLSLANG)
    set(PME_VULKAN_ON ON)
  else()
    message(STATUS "pushmpc_engine: no Vulkan engine (Vulkan-Headers: '${PME_VULKAN_HEADERS_DIR}', volk: '${PME_VOLK_DIR}', glslang: '${PME_GLSLANG}')")
  endif()
endif()

set(PME_SOURCES pmeng_module.cpp cpu_backend.cpp)
if(WIN32)
  list(APPEND PME_SOURCES d3d11_backend.cpp)
endif()
if(PME_VULKAN_ON)
  set(_g "${ROOT}/vsplugin/engines/gpu")
  add_library(pme_gpu STATIC "${_g}/vk_context.cpp" "${_g}/gpu_engine.cpp" "${_g}/subtitle_hold.cpp" "${PFP}/src/CrosshairHold.cpp" "${PME_VOLK_DIR}/volk.c")
  target_include_directories(pme_gpu PUBLIC "${_g}" "${_g}/spv" "${PME_VULKAN_HEADERS_DIR}/include" "${PME_VOLK_DIR}" "${ROOT}/vsplugin/third_party/nis")
  target_compile_definitions(pme_gpu PUBLIC PMVS_GPU VK_NO_PROTOTYPES)
  target_link_libraries(pme_gpu PUBLIC pme_portable ${CMAKE_DL_LIBS})
  set(_spv "${CMAKE_BINARY_DIR}/pme_spv")
  file(MAKE_DIRECTORY "${_spv}")
  set(_spv_out)
  foreach(s import export thumb)
    add_custom_command(OUTPUT "${_spv}/pme_${s}.h"
      COMMAND "${PME_GLSLANG}" -V --target-env vulkan1.2 --vn kSpv_pme_${s} -o "${_spv}/pme_${s}.h" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${s}.comp"
      DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${s}.comp" COMMENT "glslang: ${s}.comp" VERBATIM)
    list(APPEND _spv_out "${_spv}/pme_${s}.h")
  endforeach()
  add_custom_target(pme_spv DEPENDS ${_spv_out})
  list(APPEND PME_SOURCES vulkan_backend.cpp)
endif()
add_library(pushmpc_engine SHARED ${PME_SOURCES})
target_include_directories(pushmpc_engine PRIVATE "${ROOT}/mpvnative/include" "${ROOT}/pushmpc/include" "${ROOT}/vsplugin/src")
target_link_libraries(pushmpc_engine PRIVATE pme_portable)
if(PME_VULKAN_ON)
  add_dependencies(pushmpc_engine pme_spv)
  target_include_directories(pushmpc_engine PRIVATE "${_spv}")
  target_compile_definitions(pushmpc_engine PRIVATE PME_VULKAN)
  target_link_libraries(pushmpc_engine PRIVATE pme_gpu)
endif()
set_target_properties(pushmpc_engine PROPERTIES CXX_VISIBILITY_PRESET hidden
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/out"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out" LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/out")
if(WIN32)
  target_include_directories(pushmpc_engine PRIVATE "${ROOT}/filter/src")
  target_link_libraries(pushmpc_engine PRIVATE d3d11 dxgi d3dcompiler user32 advapi32 shell32 ole32 setupapi)
  # the stub pushcore.dll for the tests (CPU blend, runs on WARP)
  add_library(pme_pushcore_stub SHARED "${ROOT}/vsplugin/stub/pushcore_stub.cpp")
  set_target_properties(pme_pushcore_stub PROPERTIES OUTPUT_NAME pushcore
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/stub" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/stub")
  target_include_directories(pme_pushcore_stub PRIVATE "${ROOT}/pushmpc/include")
  target_link_libraries(pme_pushcore_stub PRIVATE d3d11 dxgi)
endif()

add_executable(engine_test "${ROOT}/mpvnative/tests/engine_test.cpp")
target_include_directories(engine_test PRIVATE "${ROOT}/mpvnative/include")
target_link_libraries(engine_test PRIVATE pushmpc_engine)
set_target_properties(engine_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/out")
if(WIN32)
  target_link_libraries(engine_test PRIVATE d3d11 dxgi)
  add_dependencies(engine_test pme_pushcore_stub)
  target_compile_definitions(engine_test PRIVATE PME_STUB_DIR="${CMAKE_BINARY_DIR}/stub")
endif()
add_test(NAME engine_test COMMAND engine_test)
