# pmml: PushMPC's ML runtime layer (see include/pmml/pmml.h). A static library for pushcore and the vsplugin, the
# pmml_cli tool (list devices, run a clip, probe/benchmark, Auto) and the unit tests.
#   -DPMML_ORT_INCLUDE=<dirs>  ONNX Runtime C API headers, first (a ;-list: more folders after it) (onnxruntime_c_api.h; dml_provider_factory.h + DirectML.h for
#                              DirectML on Windows). ONNX Runtime itself is loaded at run time (no import library).
#   -DPMML_NCNN_DIR=<dir>      an ncnn release package (lib/cmake/ncnn) for the ncnn-Vulkan backend (optional)
cmake_minimum_required(VERSION 3.20)
project(pmml CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# MSVC runtime: the parent's choice (the vsplugin builds /MT). ncnn's Windows release package is /MD, so the ncnn
# backend links only into a /MD build (pmml_cli on its own); the vsplugin on Windows uses ONNX Runtime only.

set(PMML_ORT_INCLUDE "" CACHE STRING "ONNX Runtime C API headers (;-list)")
set(PMML_NCNN_DIR "" CACHE PATH "ncnn release package")
set(_pmml_ort_first "")
if(PMML_ORT_INCLUDE)
  list(GET PMML_ORT_INCLUDE 0 _pmml_ort_first)
endif()
if(NOT _pmml_ort_first OR NOT EXISTS "${_pmml_ort_first}/onnxruntime_c_api.h")
  message(FATAL_ERROR "pmml: set PMML_ORT_INCLUDE to a folder with onnxruntime_c_api.h (pmml/scripts/fetch_runtime.py --headers)")
endif()

add_library(pmml STATIC src/common.cpp src/platform.cpp src/cpu.cpp src/ort.cpp src/autobench.cpp)
target_include_directories(pmml PUBLIC include PRIVATE src ${PMML_ORT_INCLUDE})
find_package(Threads REQUIRED)
target_link_libraries(pmml PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  target_compile_definitions(pmml PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(pmml PUBLIC d3d12 dxgi)
endif()
if(MSVC)
  target_compile_options(pmml PRIVATE /W3 /utf-8)
else()
  target_compile_options(pmml PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()

set(PMML_WITH_NCNN 0)
if(PMML_NCNN_DIR)
  find_package(ncnn CONFIG PATHS "${PMML_NCNN_DIR}" "${PMML_NCNN_DIR}/lib/cmake/ncnn" NO_DEFAULT_PATH)
  if(ncnn_FOUND)
    set(PMML_WITH_NCNN 1)
    target_sources(pmml PRIVATE src/ncnn_backend.cpp src/ncnn_warp.cpp)
    target_link_libraries(pmml PUBLIC ncnn)
  else()
    message(WARNING "pmml: no ncnn package in ${PMML_NCNN_DIR}: the ncnn backend is left out")
  endif()
endif()
target_compile_definitions(pmml PUBLIC PMML_WITH_NCNN=${PMML_WITH_NCNN})

add_executable(pmml_cli tools/pmml_cli.cpp tools/y4m.h)
target_link_libraries(pmml_cli PRIVATE pmml)
add_executable(pmml_unit tests/pmml_unit.cpp)
target_link_libraries(pmml_unit PRIVATE pmml)
target_include_directories(pmml_unit PRIVATE src)
enable_testing()
add_test(NAME pmml_unit COMMAND pmml_unit)
