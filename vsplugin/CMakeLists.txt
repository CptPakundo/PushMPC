# pushmpc_vs: the VapourSynth plugin pushmpc.Enhance with its backends, the stub pushcore.dll
# for tests and CI, and the tests:
#   portable_test  every OS: backend selection, the CPU budget, settings.ini and folders, the portable engine on
#                  synthetic frames (no VapourSynth, no GPU);
#   sched_test     Windows: scheduling, rates, reuse, bandwidth, Auto glue;
#   core_test      Windows: the D3D11 host and the pipeline against the stub on WARP.
# Backends: nvidia (pushcore, Windows only), apple (VideoToolbox, macOS; engines/apple) and portable (PushFPS's portable core, third_party/pushfps-portable, GPL-3)
# are compiled in; rife is a script preset in pushmpc.vpy. The plugin needs VapourSynth's SDK headers (VapourSynth4.h,
# VSHelper4.h, LGPL) in third_party/vapoursynth; without them only the stub and the tests are built.
# The same tree builds with MSVC on Windows, gcc / clang on Linux and clang on macOS:
#   cmake -S vsplugin -B vsplugin/build -DCMAKE_BUILD_TYPE=Release && cmake --build vsplugin/build && ctest --test-dir vsplugin/build

# commits engines/gpu/gpu-pins.json pins). Shaders: the checked-in SPIR-V (engines/gpu/spv), or made anew with
# -DPMVS_GLSLANG=<glslangValidator>. Without the headers the plugin is built with the CPU engine only.

    include(FetchContent)
    string(JSON _vh_url GET "${_pins}" vulkanHeaders url)
    string(JSON _volk_url GET "${_pins}" volk url)
    string(JSON _vh_sha GET "${_pins}" vulkanHeaders sha256)
    string(JSON _volk_sha GET "${_pins}" volk sha256)
    FetchContent_Declare(pm_vulkan_headers URL "${_vh_url}" URL_HASH SHA256=${_vh_sha} DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_Declare(pm_volk URL "${_volk_url}" URL_HASH SHA256=${_volk_sha} DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_GetProperties(pm_vulkan_headers)
    if(NOT pm_vulkan_headers_POPULATED)
      FetchContent_Populate(pm_vulkan_headers)
    endif()
    FetchContent_GetProperties(pm_volk)
    if(NOT pm_volk_POPULATED)
      FetchContent_Populate(pm_volk)
    endif()
    set(PMVS_VULKAN_HEADERS_DIR "${pm_vulkan_headers_SOURCE_DIR}" CACHE PATH "" FORCE)
    set(PMVS_VOLK_DIR "${pm_volk_SOURCE_DIR}" CACHE PATH "" FORCE)
  endif()
  if(EXISTS "${PMVS_VULKAN_HEADERS_DIR}/include/vulkan/vulkan_core.h" AND EXISTS "${PMVS_VOLK_DIR}/volk.c")
    set(PMVS_GPU_ON ON)

  set_target_properties(apple_vt_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
  target_link_libraries(apple_vt_test PRIVATE pmvs_apple)
  # pushmpc-iina.so: the IINA in-process bridge (an mpv C plugin that patches libmpv's libavfilter imports; see
  # src/iina_bridge.mm). Needs FFmpeg's and mpv's headers matching IINA's majors (CI: Homebrew ffmpeg / mpv); links
  # no FFmpeg or mpv itself (their symbols come from the host process).
  option(PUSHMPC_IINA_BRIDGE "IINA in-process bridge (pushmpc-iina.so)" OFF)
  if(PUSHMPC_IINA_BRIDGE)
    set(FFMPEG_INCLUDE "" CACHE PATH "FFmpeg include directory")
    set(MPV_INCLUDE "" CACHE PATH "mpv include directory (mpv/client.h)")
    # The bridge uses FFmpeg's structures, so it runs only in an app whose libavutil major is the one it was built for
    # (its guard says so otherwise). One build per FFmpeg major: pushmpc-iina.so from FFMPEG_INCLUDE (Homebrew's FFmpeg)
    # and, for every include folder in PUSHMPC_IINA_FFMPEG_INCLUDES (e.g. Homebrew's ffmpeg@7 / ffmpeg@8), one more;
    # each is also named by its libavutil major, pushmpc-iina-av<N>.so, which pushmpc-apps-mac.sh picks per app
    # (Stremio 5.1.26 bundles FFmpeg 7 = libavutil 59).
    set(PUSHMPC_IINA_FFMPEG_INCLUDES "" CACHE STRING "more FFmpeg include folders, one bridge each (pushmpc-iina-av<libavutil major>.so)")
    function(pmvs_avutil_major inc out)
      set(m "")
      foreach(h version_major.h version.h)
        if(NOT m AND EXISTS "${inc}/libavutil/${h}")
          file(STRINGS "${inc}/libavutil/${h}" l REGEX "^#define LIBAVUTIL_VERSION_MAJOR[ \t]+[0-9]+")
          if(l)
            string(REGEX REPLACE ".*LIBAVUTIL_VERSION_MAJOR[ \t]+([0-9]+).*" "\\1" m "${l}")
          endif()
        endif()
      endforeach()
      set(${out} "${m}" PARENT_SCOPE)
    endfunction()
    function(pmvs_iina_bridge tgt name inc)
      add_library(${tgt} MODULE src/iina_bridge.mm)
      set_target_properties(${tgt} PROPERTIES OUTPUT_NAME ${name} PREFIX "" SUFFIX ".so" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
      target_include_directories(${tgt} PRIVATE ${inc} ${MPV_INCLUDE} engines/apple engines/portable src ${CMAKE_CURRENT_SOURCE_DIR}/../mpvnative/include
                                 "${PUSHMPC_CORE_INCLUDE}")   # pushmpc/content_detect.h (ContentDetect, with or without the image model)
      target_compile_options(${tgt} PRIVATE -fobjc-arc -fvisibility=hidden)
      if(PMML_ORT_INCLUDE)   # ContentDetect's image model through pmml (CoreML / CPU); the pmml target is added below
        target_link_libraries(${tgt} PRIVATE pmml)
        target_compile_definitions(${tgt} PRIVATE PUSHMPC_CONTENT_MODEL=1)
      endif()
      target_link_libraries(${tgt} PRIVATE pmvs_apple pmvs_portable "-undefined dynamic_lookup")
    endfunction()
    pmvs_iina_bridge(pushmpc_iina pushmpc-iina "${FFMPEG_INCLUDE}")
    pmvs_avutil_major("${FFMPEG_INCLUDE}" _iina_main)
    if(_iina_main)   # the default build under its major's name too
      add_custom_command(TARGET pushmpc_iina POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "$<TARGET_FILE:pushmpc_iina>" "${CMAKE_BINARY_DIR}/pushmpc-iina-av${_iina_main}.so" VERBATIM)
    endif()
    # each extra major is its own target (pushmpc_iina_av<N>), never a dependency of pushmpc_iina: a major that does not
    # build cannot take the default bridge down. pushmpc_iina_extras builds them all (CI: every listed major must build);
    # the formula builds them one by one and skips a failing one. Their majors: iina-bridge-majors.txt in the build folder.
    set(_iina_done "${_iina_main}")
    set(_iina_extra "")
    add_custom_target(pushmpc_iina_extras)
    foreach(inc IN LISTS PUSHMPC_IINA_FFMPEG_INCLUDES)
      pmvs_avutil_major("${inc}" m)
      if(NOT m)
        message(FATAL_ERROR "IINA bridge: no libavutil version in ${inc}")
      endif()
      if(m IN_LIST _iina_done)
        continue()
      endif()
      list(APPEND _iina_done ${m})
      list(APPEND _iina_extra ${m})
      pmvs_iina_bridge(pushmpc_iina_av${m} pushmpc-iina-av${m} "${inc}")
      add_dependencies(pushmpc_iina_extras pushmpc_iina_av${m})
    endforeach()
    string(REPLACE ";" "\n" _iina_lines "${_iina_extra}")
    file(WRITE "${CMAKE_BINARY_DIR}/iina-bridge-majors.txt" "${_iina_lines}\n")
    message(STATUS "IINA bridge: builds for libavutil ${_iina_done} (default ${_iina_main}; targets pushmpc_iina, pushmpc_iina_extras)")
  endif()
endif()
